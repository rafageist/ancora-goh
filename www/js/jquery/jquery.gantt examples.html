Manipulation

// Add Task
$('#gantt01').addTask({ 'titles': 'Task04', 'start_date': '20100529', 'end_date': '20100610' }); [test]
$('#gantt01').addTask({
	'titles': 'Task05',
	'start_date': '20100602',
	'end_date': '20100615',
	'color': '#0000FF',
	'text': 'Task05'
}); [test]

// Today
$('#gantt01').setPeriod();

// Next Page
$('#gantt01').setPeriod('+');

// +7th
$('#gantt01').setPeriod('+7');

// Prev Page
$('#gantt01').setPeriod('-');

// -7th
$('#gantt01').setPeriod('-7');
